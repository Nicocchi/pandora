121/* Declare a target attribute table called NAME that only has GNU attributes.
122 There should be no null trailing element. E.g.:
123
124 TARGET_GNU_ATTRIBUTES (aarch64_attribute_table,
125 {
126 { "aarch64_vector_pcs", ... },
127 ...
128 }); */
129
130#define TARGET_GNU_ATTRIBUTES(NAME, ...) \
131 static const attribute_spec NAME##_2[] = __VA_ARGS__; \
132 static const scoped_attribute_specs NAME##_1 = { "gnu", { NAME##_2 } }; \
133 static const scoped_attribute_specs *const NAME[] = { &NAME##_1 }
