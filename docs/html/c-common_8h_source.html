920/* Return true if division or modulo op0 / op1 or op0 % op1 may be shortened.
921 We can shorten only if we can guarantee that op0 is not signed integral
922 minimum or op1 is not -1, because e.g. (long long) INT_MIN / -1 is

928 tree type0 = TREE_TYPE (op0);
929 if (TYPE_UNSIGNED (type0))
930 return true;
931 /* A cast from narrower unsigned won't be signed integral minimum,
932 but cast from same or wider precision unsigned could be. */
933 if (TREE_CODE (op0) == NOP_EXPR
934 && INTEGRAL_TYPE_P (TREE_TYPE (TREE_OPERAND (op0, 0)))
935 && TYPE_UNSIGNED (TREE_TYPE (TREE_OPERAND (op0, 0)))
936 && (TYPE_PRECISION (TREE_TYPE (TREE_OPERAND (op0, 0)))
937 < TYPE_PRECISION (type0)))
938 return true;
939 if (TREE_CODE (op1) == INTEGER_CST && !integer_all_onesp (op1))
940 return true;
941 return false;
942}
