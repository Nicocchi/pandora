120inline bool
121should_remove_lhs_p (tree lhs)
122{
123 return (lhs
124 && TREE_CODE (TYPE_SIZE_UNIT (TREE_TYPE (lhs))) == INTEGER_CST
125 && !TREE_ADDRESSABLE (TREE_TYPE (lhs)));
126}
