38/* How to output an ASCII string constant. */
39
40#undef ASM_OUTPUT_ASCII
41#define ASM_OUTPUT_ASCII(FILE, PTR, SIZE) \
42do \
43{ size_t i = 0, limit = (SIZE); \
44 while (i < limit) \
45 { if (i%10 == 0) { if (i!=0) putc ('\n', (FILE)); \
46 fputs (ASM_BYTE, (FILE)); } \
47 else putc (',', (FILE)); \
48 fprintf ((FILE), "0x%x", ((PTR)[i++] & 0377)) ;} \
49 putc ('\n', (FILE)); \
50} while (0)
