93inline wide_int_bitmask
94wide_int_bitmask::operator << (int amount)
95{
96 wide_int_bitmask ret;
97 if (amount >= 64)
98 {
99 ret.low = 0;
100 ret.high = low << (amount - 64);
101 }
102 else if (amount == 0)
103 ret = *this;

105 {
106 ret.low = low << amount;
